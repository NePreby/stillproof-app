// Generator for these fixtures, run against stillproof feat/embed (4197c53)
// from a module that replaces github.com/stillproof-labs/stillproof with it.
// It writes original.jpg and receipt.json. Then, in the stillproof repo:
//
//	go run ./cmd/stillproof embed --key SIGNER original.jpg receipt.json > embedded.jpg
//	go run ./cmd/stillproof embed --key SIGNER original.jpg receipt-attested.json > embedded-attested.jpg
//
// receipt-attested.json is receipt.json plus an android-key attestation of two
// 26000-byte dummy certificates (bytes i % 251, base64), so the store spans two
// APP11 segments. SIGNER is any P-256 key in hex. The anchor tx is made up.
package main

import (
	"bytes"
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"image"
	"image/color"
	"image/jpeg"
	"os"
	"time"

	"github.com/stillproof-labs/stillproof/claim"
	"github.com/stillproof-labs/stillproof/merkle"
	"github.com/stillproof-labs/stillproof/receipt"
)

func main() {
	img := image.NewRGBA(image.Rect(0, 0, 32, 24))
	for y := 0; y < 24; y++ {
		for x := 0; x < 32; x++ {
			img.Set(x, y, color.RGBA{uint8(x * 8), uint8(y * 10), 90, 255})
		}
	}
	var buf bytes.Buffer
	if err := jpeg.Encode(&buf, img, &jpeg.Options{Quality: 80}); err != nil {
		panic(err)
	}
	b := buf.Bytes()
	app0 := []byte{0xFF, 0xE0, 0x00, 0x10, 'J', 'F', 'I', 'F', 0, 1, 1, 0, 0, 1, 0, 1, 0, 0}
	photo := append(append(append([]byte{}, b[:2]...), app0...), b[2:]...)

	d, _ := hex.DecodeString("4b3de239f32b1c6fd428457f964729d5d9390adef02ca76f0bd3ef01eff1e74c")
	key, err := ecdsa.ParseRawPrivateKey(elliptic.P256(), d)
	if err != nil {
		panic(err)
	}
	s, err := claim.Sign(claim.Claim{SHA256: sha256.Sum256(photo), Type: "image/jpeg", Captured: 1762315692000, Key: &key.PublicKey}, key)
	if err != nil {
		panic(err)
	}
	leaves := []merkle.Hash{claim.LeafHash(s.Bytes()), sha256.Sum256([]byte("another leaf"))}
	root := merkle.Root(leaves)
	var proof []string
	for _, p := range merkle.Proof(leaves, 0) {
		proof = append(proof, hex.EncodeToString(p[:]))
	}
	r := receipt.Receipt{
		V: 1, Claim: string(s.Bytes()), Index: 0, Size: 2, Proof: proof, Root: hex.EncodeToString(root[:]),
		Anchor: receipt.Anchor{
			Network: "testnet", Account: "GAAZI4TCR3TY5OJHCTJC2A4QSY6CJWJH5IAJTGKIN2ER7LBNVKOCCWN7",
			Tx: hex.EncodeToString(bytes.Repeat([]byte{0xab}, 32)), Ledger: 5565275,
			ClosedAt: time.Date(2026, 11, 5, 4, 8, 12, 0, time.UTC),
		},
	}
	rj, err := json.Marshal(r)
	if err != nil {
		panic(err)
	}
	must(os.WriteFile("original.jpg", photo, 0o644))
	must(os.WriteFile("receipt.json", rj, 0o644))
}

func must(err error) {
	if err != nil {
		panic(err)
	}
}
